% ---- Accent bar (left strip inside a named stage node) ----
\newcommand{\accentbar}[2]{%
  \fill[#2] ([xshift=0.55mm,yshift=-0.9mm]#1.north west)
    rectangle ([xshift=1.95mm,yshift=0.9mm]#1.south west);}

% Soft role wash (top band) so stage type reads without relying on blue alone.
\newcommand{\rolewash}[2]{%
  \fill[#2, opacity=0.12] ([xshift=1.95mm,yshift=-0.65mm]#1.north west)
    rectangle ([xshift=-0.65mm,yshift=-7.2mm]#1.north east);}
