% Standalone figure: Fig. 1 method-overview schematic -- the coverage-debt

% ---- Accent bar (left strip inside a named stage node) ----
\newcommand{\accentbar}[2]{%
  \fill[#2] ([xshift=0.5mm,yshift=-0.85mm]#1.north west)
    rectangle ([xshift=2.0mm,yshift=0.85mm]#1.south west);}

% Soft role wash (top band) so stage type reads without relying on blue alone.
\newcommand{\rolewash}[2]{%
  \fill[#2, opacity=0.12] ([xshift=2.0mm,yshift=-0.5mm]#1.north west)
    rectangle ([xshift=-0.5mm,yshift=-7.0mm]#1.north east);}
