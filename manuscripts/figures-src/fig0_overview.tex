% ---- Accent bar (left strip inside a named stage node) ----
\newcommand{\accentbar}[2]{%
  \fill[#2] ([xshift=0.5mm,yshift=-0.8mm]#1.north west)
    rectangle ([xshift=1.8mm,yshift=0.8mm]#1.south west);}

% Soft role wash (top band) so stage type reads without relying on blue alone.
\newcommand{\rolewash}[2]{%
  \fill[#2, opacity=0.12] ([xshift=1.8mm,yshift=-0.5mm]#1.north west)
    rectangle ([xshift=-0.5mm,yshift=-6.3mm]#1.north east);}
